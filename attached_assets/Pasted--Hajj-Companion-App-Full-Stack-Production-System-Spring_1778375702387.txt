### personal_places  ← NEW FEATURE
  id                UUID PRIMARY KEY DEFAULT gen_random_uuid()
  user_id           UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE
  name_ar           VARCHAR(100) NOT NULL     -- e.g. "فندقي", "خيمة أخي"
  name_en           VARCHAR(100)
  emoji             VARCHAR(10) DEFAULT '📍'  -- user-chosen icon
  latitude          DECIMAL(10,8) NOT NULL
  longitude         DECIMAL(11,8) NOT NULL
  description       TEXT                      -- optional notes
  color             VARCHAR(7) DEFAULT '#1D9E75'  -- hex color for map pin
  is_shared         BOOLEAN DEFAULT false     -- share with group members

  created_at        TIMESTAMP DEFAULT CURRENT_TIMESTAMP
  updated_at        TIMESTAMP DEFAULT CURRENT_TIMESTAMP
  INDEX idx_personal_places_user (user_id)
  INDEX idx_personal_places_shared (shared_with_group) WHERE is_shared = true

  id                UUID PRIMARY KEY DEFAULT gen_random_uuid()
  user_id           UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE
  destination_place_id UUID REFERENCES personal_places(id) ON DELETE SET NULL

  destination_lng   DECIMAL(11,8) NOT NULL
  destination_name  VARCHAR(100) NOT NULL
  started_at        TIMESTAMP DEFAULT CURRENT_TIMESTAMP
  completed_at      TIMESTAMP                 -- null if still navigating
  cancelled_at      TIMESTAMP
  distance_total_m  FLOAT                     -- total distance at start
  distance_walked_m FLOAT DEFAULT 0           -- updated during nav
  -- INDEX on (user_id, started_at DESC)

### location_history
  id                UUID PRIMARY KEY DEFAULT gen_random_uuid()
  user_id           UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE

  is_offline_sync   BOOLEAN DEFAULT false

### emergency_alerts

### Navigation Logic (navigation_service.dart)
  // Calculate bearing between two GPS points (offline, pure math)
  double calculateBearing(double lat1, double lng1, double lat2, double lng2) {
    final dLng = (lng2 - lng1) * pi / 180;
    final lat1Rad = lat1 * pi / 180;
    final lat2Rad = lat2 * pi / 180;
    final y = sin(dLng) * cos(lat2Rad);
    final x = cos(lat1Rad) * sin(lat2Rad) - sin(lat1Rad) * cos(lat2Rad) * cos(dLng);
    return (atan2(y, x) * 180 / pi + 360) % 360; // 0-360 degrees
  }

  // Convert bearing to Arabic direction
  String bearingToArabic(double bearing) {
    if (bearing < 22.5 || bearing >= 337.5) return 'اتجه شمالاً ↑';
    if (bearing < 67.5)  return 'اتجه شمال شرقاً ↗';
    if (bearing < 112.5) return 'اتجه شرقاً →';
    if (bearing < 157.5) return 'اتجه جنوب شرقاً ↘';
    if (bearing < 202.5) return 'اتجه جنوباً ↓';
    if (bearing < 247.5) return 'اتجه جنوب غرباً ↙';
    if (bearing < 292.5) return 'اتجه غرباً ←';
    return 'اتجه شمال غرباً ↖';
  }

  // ETA calculation
  int etaMinutes(double distanceMeters) {
    const walkingSpeedMs = 1.1; // avg pilgrim walking speed m/s (slower than normal)
    return (distanceMeters / walkingSpeedMs / 60).ceil();
  }

## 7. OFFLINE HANDLING FOR PERSONAL PLACES

  Personal places are stored locally in SQLite (Flutter) / IndexedDB (Web).
  Sync strategy:
    - CREATE: write to local DB immediately, queue POST /places for sync
    - UPDATE: same — local first, queue PUT /places/{id}
    - DELETE: local delete immediately, queue DELETE /places/{id}
    - NAVIGATION: 100% offline — all math done on device (Haversine + bearing)
    - No internet required at any point for navigation!

  Local SQLite table (mirrors server):
    CREATE TABLE personal_places (
      id TEXT PRIMARY KEY,
      name_ar TEXT NOT NULL,
      name_en TEXT,
      emoji TEXT DEFAULT '📍',
      latitude REAL NOT NULL,
      longitude REAL NOT NULL,
      description TEXT,
      color TEXT DEFAULT '#1D9E75',
      is_shared INTEGER DEFAULT 0,
      visit_count INTEGER DEFAULT 0,
      last_visited_at INTEGER,   -- unix timestamp
      created_at INTEGER,
      sync_status TEXT DEFAULT 'pending'  -- pending | synced | conflict
    );

## 8. UX FLOW — HOW THE FEATURE WORKS (End-to-End)

  SAVING A PLACE:
  1. User arrives at hotel → opens app → taps "+" in Places tab
  2. Map appears centered on their GPS position
  3. They drag pin to exact hotel entrance
  4. Type "فندقي" → pick 🏨 emoji → pick green color → Save
  5. Place saved locally (offline) + synced to server when online
  6. Appears in their list with "0 م" distance (they're there now)

  NAVIGATING BACK:
  1. User is in Mina, wants to return to hotel
  2. Opens "مواقعي" tab → sees "فندقي 🏨 — 2.3 كم"
  3. Taps "ابدأ الملاحة"
  4. Compass screen opens:
      - Large arrow pointing toward hotel (updates every 3s)
      - "2.3 كم — ~29 دقيقة مشياً"
      - Mini map showing their position moving toward the hotel pin
  5. As they walk: distance updates in real-time
  6. At 100m: arrow glows green, "اقتربت! 100 م"
  7. At 30m: "وصلت إلى فندقي! 🎉" + vibration
